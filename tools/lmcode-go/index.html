lmcode-go

lmcode-go is a standard-library-only Go 1.26+ SDK (package lmcode) for verifying and executing lmcode DSLB v6 bytecode in static Go binaries. It builds cleanly with CGO_ENABLED=0 and provides the 226-signature lmcode_std_v3 catalog, rest.Controller controller, dry-mode callbacks, and LMCODE_RETURN framing.

Quick reference

git clone https://github.com/lmctl-ai/lmcode-go.git
cd lmcode-go
CGO_ENABLED=0 go test ./...
go run ./cmd/conformance conformance/vectors

lmcode.InvocationResultsFromCommand(files)(*lmcode.InvocationResults, error)Manage multi-part ResultSinks, dry mode, and LMCODE_RETURN_BEGIN/END output
package main

import (
	"os"
	lmcode "github.com/lmctl-ai/lmcode-go"
	"github.com/lmctl-ai/lmcode-go/rest"
)

func main() {
	reg := lmcode.NewRegistry()
	srv, _ := rest.Register(reg, rest.Options{MaxBodyBytes: 1 << 20})
	bc, _ := os.ReadFile(os.Args[1])
	schema, _ := os.ReadFile(os.Args[2])
	prog, err := lmcode.Load(bc, schema, reg)
	lmcode.ExitIfError("load", err)
	err = srv.Run(prog, lmcode.NewContext())
	lmcode.ExitIfError("run", err)
}
