lmdebug

lmdebug is a Linux network, system, process, packet-capture, and detection introspection CLI and LMScript host in one self-contained executable. It emits structured JSON on the command line, exposes typed domain contexts (lmdebug, net, sys, proc, detect, shark), and enforces an audited dry-run-by-default mutation framework.

npm install -g @lmctl-ai/lmctl@latest
lmdebug --version

p = _lmcode_ cli="lmdebug" install="github:lmctl-ai/lmdebug@latest";
let ifs = net.interfaces();
assert(ifs.len() > 0);

@"/sockets/listening"
fn get_listening(x: RestExchange) -> void implements RestGet {
  let sum = net.socket_summary("tcp");
  x.json("{\"listening\":" + str(sum.listening()) + "}");
}

rest_config("session", "global");
let out = rest_cli(str_split("/sockets/listening", " "));
lmcode_return_put("listening_tcp", s.listening());

end _lmcode_
res = run_lmcode(p);
require(res.ok, json_stringify(res));
