lmcode-java

lmcode-java is a zero-dependency Java 21 SDK (package io.lmctl.lmcode) for running lmcode DSLB v6 bytecode on the JVM or GraalVM Native Image. It preserves raw bytes in VmString, installs the 226-signature lmcode_std_v3 catalog, and includes the RestController and InvocationResults APIs.

Quick reference

git clone https://github.com/lmctl-ai/lmcode-java.git
cd lmcode-java
mvn -q verify
./scripts/build-native.sh

import io.lmctl.lmcode.*;
import io.lmctl.lmcode.rest.*;
import java.nio.file.*;

public final class HostMain {
    public static void main(String[] args) throws Exception {
        Registry reg = new Registry();
        RestController rest = Rest.registerRest(reg, new RestOptions().maxBodyBytes(1 << 20));
        byte[] bc = Files.readAllBytes(Path.of(args[0]));
        byte[] schema = Files.readAllBytes(Path.of(args[1]));
        Program prog = Program.load(bc, schema, reg);
        rest.run(prog, new Context());
    }
}
