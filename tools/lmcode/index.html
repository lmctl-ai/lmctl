lmcode

lmcode is the specification, DSLB v6 bytecode contract, 226-signature standard catalog, and conformance suite for schema-bound host programs. It defines the grammar, schema format, verifier rules, standard REST and CLI interfaces, invocation result protocol, and cross-runtime test vectors.

Quick reference

git clone https://github.com/lmctl-ai/lmcode.git
cd lmcode
npm install -g @lmctl-ai/lmctl
python3 tests/grammar/tree_sitter_differential.py --json /tmp/tree-sitter-diff.json

program = _lmcode_ schema="host.schema.json";

@"/users/:id"
fn get_user(x: RestExchange) -> void implements RestGet {
  x.json("{\"id\":\"" + x.param("id") + "\"}");
}

@"status"
fn cli_status(x: RestExchange) -> void implements CliCommand {
  x.send("ok:" + x.param("arg0"));
}

rest_config("prefix", "/api");
$return.preview = rest_cli(qw/status ready/);
rest_serve();
end _lmcode_
