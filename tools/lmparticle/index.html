lmparticle

lmparticle is a seeded Monte Carlo photon transport host for LMScript with embedded NIST XCOM cross sections. It simulates shielding, detector spectra, radiographic contrast, and collision kerma from 1 keV to 20 MeV in a self-contained 2.5 MB executable.

Quick reference

npm install -g @lmctl-ai/lmctl
lmparticle --version
lmparticle --lmcode-schema --json
lmctl autorun examples/depth_dose.lms

p = _lmcode_ cli="lmparticle" install="github:lmctl-ai/lmparticle@latest";
let pb = rad.builtin("lead");
let g = rad.geometry().slab("shield", pb, 0.0, 1.0);
let s = rad.source().point(0.0, 0.0, -1.0).beam(0.0, 0.0, 1.0).energy(0.662);
let r = rad.simulation(g, s).plane("exit", "z", 1.0).seed(1).run(200000);
rad.assert_close(r.plane_uncollided("exit"), pow(2.718281828459045, -pb.mu(0.662)), 0.005);
end _lmcode_
r = run_lmcode(p);
require(r.ok, json_stringify(r));
