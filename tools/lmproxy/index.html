lmproxy

lmproxy is a statically-linked HTTP/HTTPS proxy daemon, URL and HTTP/1.x parser CLI, and LMScript host built on the embedded Squid 7 engine. It runs forward and transparent interception proxies with SSL-bump TLS inspection alongside structured URL and message parsing in _lmcode_.

Quick reference

npm install -g @lmctl-ai/lmctl@latest
lmproxy --version
lmproxy --lmcode-schema
lmproxy --lmcode-signatures

p = _lmcode_ cli="lmproxy" install="github:lmctl-ai/lmproxy@latest";
let u = proxy.parse_url("http://user@example.com:8080/a/b?x=1");
assert_eq(u.valid(), true);
assert_eq(u.host(), "example.com");
assert_eq(u.port(), 8080);

@"/request/inspect"
fn get_inspect(x: RestExchange) -> void implements RestGet {
  let m = proxy.parse_request("GET /status HTTP/1.1\r\nHost: proxy.local\r\n\r\n");
  x.json("{\"method\":\"" + m.method() + "\",\"host\":\"" + m.header("host") + "\"}");
}
assert_eq(rest_cli(str_split("/request/inspect", " ")), "{\"method\":\"GET\",\"host\":\"proxy.local\"}");
end _lmcode_
res = run_lmcode(p);
require(res.ok, json_stringify(res));
