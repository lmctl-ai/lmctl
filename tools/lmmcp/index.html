lmmcp

lmmcp is a pure-Go Model Context Protocol (MCP) bridge and LMScript host exposing stdio, Streamable HTTP, and HTTP+SSE MCP servers over REST, CLI, and JSON-lines stdio. It provides typed _lmcode_ functions (mcp_server, mcp_call, mcp_read) alongside the 226 standard lmcode_std_v3 REST and CLI interfaces.

Quick reference

git clone https://github.com/lmctl-ai/lmmcp.git
cd lmmcp && CGO_ENABLED=0 go build -o lmmcp ./cmd/lmmcp
./lmmcp --version
./lmmcp --lmcode-schema
./lmmcp --lmcode-signatures

Standard REST and CLIRestGet..RestOptions, RestHandler, CliCommand, CliOptionExpose _lmcode_ functions as HTTP routes (:param) or stateless CLI flags and commands via rest_serve and rest_cli.
 
p = _lmcode_ schema="examples/schema.json";
let srv = mcp_server("fixture");
let tools = srv.tools();
let reply = srv.call("echo", json_parse("{\"text\":\"hello from lmcode\"}"));

@"/call/:tool"
fn get_call(x: RestExchange) -> void implements RestGet {
  let res = mcp_call("fixture", x.param("tool"), json_parse("{\"text\":\"hi\"}"));
  x.json(res.to_json());
}

rest_config("session", "global");
let route_out = rest_cli(str_split("/call/echo", " "));
$return = json_parse("{\"direct\":" + reply.to_json() + ",\"route\":" + route_out + "}");
end _lmcode_
r = run_lmcode(p, {command: ["lmmcp", "--config", ARGV[0], "--lmcode", "$DSLFILE"]});
require(r.ok, json_stringify(r));
