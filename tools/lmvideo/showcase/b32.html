# B32 / P.7: original orbital accretion scene plus Image grading, projected HUD and live Plot/quiver telemetry.
mkdir("build");mkdir("build/bench-32-frames"); program=_lmcode_ cli="lmvideo";
video.size(1280,720);video.fps(30);video.frames(150);video.crf(20);video.output("build/bench-32-multidomain-orbit-video.mp4");video.begin();
let scene=scene3d().threads(4).background("#050914").ambient(0.02).tonemap("aces",1.0);scene.camera(camera3d().look_at(0.0,2.4,7.2,0.0,0.4,0.0).perspective(38.0,0.1,40.0));scene.directional_light(-2.0,4.0,3.0,"#9fc8ff",1.2);
let star=scene.add(mesh_sphere(0.55,24,40),material("#f2c86b",0.0,0.35).emissive("#ffb347",1.5));let planet=scene.add(mesh_sphere(0.28,20,32),material("#5aa2c8",0.0,0.45));let moon=scene.add(mesh_sphere(0.10,16,24),material("#d1d8e0",0.0,0.7));let accretion=point_cloud();for(k in range(0,720)){let a=float(k)*2.399963;let r=0.9+0.003*float(k%180);accretion.add(r*cos(a),0.02*sin(a*3.0),r*sin(a),0.018,"#d8884a",0.75,true);}scene.points(accretion);
let plot=canvas.plot(920.0,430.0,310.0,240.0).title("orbital phase space").xlabel("x").ylabel("v").x_range(-1.2,1.2).y_range(-1.2,1.2).background("#f5f7fa");let orbit=plot.series("phase","line","#8f4bd6").width(2.0);let field=plot.quiver("velocity").color("#374151").scale(0.15);for(i in range(0,7)){for(j in range(0,5)){let x=-1.0+float(i)*0.33;let y=-0.8+float(j)*0.4;field.add(x,y,-y*0.2,x*0.2);}}
let lut=image_hald(4).hsl(10.0,1.12,1.0).levels(0.03,1.05,0.96);let destination=image_create(1280,720,0.0,0.0,0.0,1.0);let bloom=image_create(1280,720,0.0,0.0,0.0,1.0);let telemetry=image_create(1280,720,0.0,0.0,0.0,0.0);let composited=image_create(1280,720,0.0,0.0,0.0,1.0);let layout=layout_create(1280,720);let camera_projection=scene.projection(1280,720);
for(frame in range(0,video.count())){video.frame();let t=video.t();let phase=t*6.283185307179586/5.0;let px=1.65*cos(phase);let py=0.4;let pz=1.65*sin(phase);planet.position(px,py,pz);moon.position(px+0.42*cos(phase*4.0),py,pz+0.42*sin(phase*4.0));scene.render_image(1280,720,destination);bloom.blend(destination,"over",1.0,0,0);bloom.blur(0.0,8.0);destination.clut(lut);destination.blend(bloom,"screen",0.22,0,0);orbit.add(cos(phase),sin(phase));let projected=camera_projection.at(px,py,pz);canvas.clear("#00000000");canvas.image(destination,0.0,0.0);if(projected.visible()){canvas.text("planet",projected.x()+14.0,projected.y(),font(18.0),paint().color("#ffffff"));canvas.line(projected.x(),projected.y(),projected.x()+12.0,projected.y(),paint().stroke().stroke_width(2.0).color("#ffffff"));}canvas.text("B32 original accretion/orbit · Image bloom/Hald LUT · projected telemetry",24.0,34.0,font(22.0),paint().color("#ffffff"));plot.draw();canvas.snapshot(telemetry);layout.clear().add(destination,"over",1.0,0,0).add(telemetry,"over",1.0,0,0);layout.render(composited);canvas.clear("#050914");canvas.image(composited,0.0,0.0);video.save_frame("build/bench-32-frames/frame-"+str(video.index())+".png");if(video.index()==video.count()-1){composited.save("build/bench-32-multidomain-orbit-video-poster.png");let overview=image_pattern(900,560,"0.01","0.02","0.04","1");overview.blend(composited.resize(900,506),"over",1.0,0,27);overview.save("build/bench-32-multidomain-orbit-video.png");}}
video.end();let hero=scene3d().threads(4).background("#050914").ambient(0.0).tonemap("aces",1.0);hero.camera(camera3d().look_at(0.0,2.4,7.2,0.0,0.4,0.0).perspective(38.0,0.1,40.0));hero.directional_light(-2.0,4.0,3.0,"#9fc8ff",1.2);hero.add(mesh_sphere(0.55,24,40),material("#f2c86b",0.0,0.35).emissive("#ffb347",1.5));hero.add(mesh_sphere(0.28,20,32),material("#5aa2c8",0.0,0.45)).position(1.65,0.4,0.0);hero.render_image(900,560,"cycles",64,6,20261105,true).save("build/bench-32-cycles-hero.png");out.print("B32 original multi-domain 3D+Image+HUD+Plot/VectorField | 150 frames | separate Cycles hero still");
end _lmcode_
savedsl(program,"build/bench-32-multidomain-orbit-video.dslb");r=system(concat(dsl_cli(program),["--dsl","build/bench-32-multidomain-orbit-video.dslb"]),900000);assert(r.code==0,r.stdout+r.stderr);println(r.stdout);
