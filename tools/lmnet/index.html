lmnet

lmnet is a statically-linked Linux network diagnostics CLI and LMScript host combining iputils ping and tracepath, iperf3, rtnetlink, procfs, DNS, and TCP scanning. It provides both direct JSON command-line introspection and strongly-typed _lmcode_ scripting via the net context and standard REST and CLI interfaces.

lmnet --version

p = _lmcode_ cli="lmnet" install="github:lmctl-ai/lmnet@latest";
let links = net.links();
assert(links.len() > 0);
let addrs = net.dns_lookup("localhost");
assert(addrs.len() > 0);

@"--dev"
fn opt_dev(x: RestExchange) -> void implements CliOption {
  x.set("dev", x.query("dev"));
}

@"link-check"
fn cmd_link(x: RestExchange) -> void implements CliCommand {
  let target = x.get("dev");
  for (i in range(0, links.len())) {
    if (links.at(i).name() == target) {
      x.send(target + ":up=" + str(links.at(i).up()));
    }
  }
}

rest_config("session", "global");
assert_eq(rest_cli(str_split("--dev=lo link-check", " ")), "lo:up=true");
end _lmcode_
res = run_lmcode(p);
require(res.ok, json_stringify(res));
