lminspect

lminspect is a pure-Go HTTP/HTTPS proxy, dynamic TLS MITM interceptor, flow recorder, and LMScript host modeled on mitmproxy. It runs bytecode-compiled _lmcode_ addons inside the proxy pipeline, enforces mandatory credential sanitization on all outputs, and supports standard REST and CLI interfaces.

Quick reference

npm install -g @lmctl-ai/lmctl@latest
lminspect --version
lminspect --lmcode-schema
lminspect --lmcode-signatures

p = _lmcode_ cli="lminspect" install="github:lmctl-ai/lminspect@latest";
@request
fn on_req(f: Flow) -> void implements RequestHook {
  f.request().set_header("X-Proxy-By", "lminspect");
}

@"/flow/:seg"
fn get_flow(x: RestExchange) -> void implements RestGet {
  let f = inspect.new_flow("POST", "https://api.internal/" + x.param("seg"));
  f.respond(201, "{\"seg\":\"" + x.param("seg") + "\"}", "application/json");
  x.json("{\"status\":" + str(f.response().status_code()) + ",\"path\":\"" + f.request().path() + "\"}");
}

rest_config("session", "global");
let out = rest_cli(str_split("/flow/orders", " "));
assert(out == "{\"status\":201,\"path\":\"/orders\"}", "unexpected rest_cli output");
end _lmcode_
res = run_lmcode(p);
require(res.ok, json_stringify(res));
