lmssl

lmssl is a static executable bundling the OpenSSL 4.0.3 CLI and embedded openssl.cnf with an LMScript bytecode VM for X.509, PKI, TLS, and crypto scripting. It supports full OpenSSL subcommands, step-by-step TLS handshake tracing, AEAD and key primitives, and standard lmcode_std_v3 REST and CLI interfaces.

p = _lmcode_ cli="lmssl" install="github:lmctl-ai/lmssl@latest";
let ca_key = ssl.gen_ec("P-256");
let ca = ssl.new_cert(ca_key, "CN=Example Root CA").ca(true).valid_days(365).sign_self(ca_key, "SHA256");
let leaf_key = ssl.gen_ed25519();
let leaf = ssl.new_cert(leaf_key, "CN=*.example.test").san("DNS:*.example.test,DNS:example.test").valid_days(90).sign_with(ca, ca_key, "SHA256");
assert_eq(leaf.verify_with(ca), true);
assert_eq(leaf.matches_host("api.example.test"), true);

@"/cert/subject"
fn get_subject(x: RestExchange) -> void implements RestGet {
  x.json("{\"subject\":\"" + leaf.subject() + "\",\"ca\":" + (ca.is_ca() ? "true" : "false") + "}");
}
assert_eq(rest_cli(str_split("/cert/subject", " ")), "{\"subject\":\"CN=*.example.test\",\"ca\":true}");
end _lmcode_
