A seeded 128x128 FBM-like heightfield is colored by elevation/slope and paired with a top-down 2D marching-contour heatmap panel.

Locally verified acceptance capture. This source candidate and its required renderer are not yet a public release.

Source

# B09: original seeded four-octave Perlin FBM; the same samples drive mesh and contours.
mkdir("build");p=_lmcode_ cli="lmvideo";
let heights=texture_noise("perlin",14159,3).mix(texture_noise("perlin",2718,6),0.35).mix(texture_noise("perlin",31415,12),0.18).mix(texture_noise("perlin",16180,24),0.08).bake(128,128,"linear").image(0);
let terrain=mesh_heightfield(heights,5.0,5.0,2.0);let scalars="";
for(j in range(0,128)){for(i in range(0,128)){let left=i-1;if(left<0){left=0;}let right=i+1;if(right>127){right=127;}let top=j-1;if(top<0){top=0;}let bottom=j+1;if(bottom>127){bottom=127;}let dx=(heights.pixel(right,j,0)-heights.pixel(left,j,0))*25.4;let dz=(heights.pixel(i,bottom,0)-heights.pixel(i,top,0))*25.4;let slope=sqrt(dx*dx+dz*dz);if(slope>1.0){slope=1.0;}let value=0.82*heights.pixel(i,j,0)+0.18*slope;if(scalars!=""){scalars=scalars+",";}scalars=scalars+str(value);}}
terrain.set_vertex_scalars(scalars,"viridis",0.0,1.0);image_assert_true(terrain.vertices()==16384&&terrain.triangles()==32258);
let scene=scene3d().threads(4).background("#edf3f8").ambient(0.18);scene.camera(camera3d().look_at(5.0,4.0,7.0,0.0,0.75,0.0).perspective(42.0,0.1,100.0));scene.add(terrain,material("#ffffff",0.0,0.85));scene.directional_light(1.0,4.0,3.0,"#ffffff",1.5);let rendered=scene.render_image(560,460,2);
video.size(900,560);video.frames(1);video.output("build/bench-09-terrain-contours.png");video.begin();video.frame();canvas.clear("#edf3f8");canvas.image(rendered,20.0,60.0);
let plot=canvas.plot(600.0,60.0,280.0,450.0).title("Same FBM samples").xlabel("x / m").ylabel("z / m").x_range(-2.5,2.5).y_range(-2.5,2.5).legend("none").background("#ffffff");let contours=plot.heatmap("elevation",33,33).palette("viridis").range(0.0,1.0).contours(10,"#17202a",1.0).colorbar(true);
for(j in range(0,33)){for(i in range(0,33)){contours.cell(32-j,i,heights.pixel(i*127/32,j*127/32,0));}}plot.draw();canvas.text("128 x 128 seeded FBM | elevation + slope vertex colour",20.0,35.0,font(19.0),paint().color("#17364d"));video.end();out.print("16384 vertices; 32258 triangles; four pinned Perlin seeds; identical contour source.");
end _lmcode_
savedsl(p,"build/bench-09-terrain-contours.dslb");r=system(concat(dsl_cli(p),["--dsl","build/bench-09-terrain-contours.dslb"]),600000);assert(r.code==0,r.stderr+r.stdout);println(r.stdout);
