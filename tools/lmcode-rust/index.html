lmcode-rust

lmcode-rust is a zero-dependency Rust SDK (dslvm crate) for loading, verifying, and executing lmcode DSLB v6 bytecode in native Rust hosts. It ships the 226-signature lmcode_std_v3 catalog, the stdio REST and stateless CLI controller, dry-mode callback slots, and the LMCODE_RETURN invocation collector.

Quick reference

git clone https://github.com/lmctl-ai/lmcode-rust.git
cd lmcode-rust
cargo test

use dslvm::rest::{register_rest, RestOptions};
use dslvm::{Context, Program, Registry, Value};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut registry = Registry::new();
    registry.function("ping", &[], "string", |_ctx, _args| Ok(Value::string("pong")))?;
    let mut rest = register_rest(&mut registry, RestOptions::default())?;
    let schema = include_bytes!("../std/lmcode-std.schema.json");
    let bytecode = std::fs::read("app.dslb")?;
    let program = Program::load(&bytecode, schema, &registry)?;
    let mut ctx = Context::new();
    rest.run(&program, &registry, &mut ctx)?;
    Ok(())
}
